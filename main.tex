\section{Why GPU-native modeling, only for NLPs?}
% =============================================================
\begin{frame}{Why a GPU-compatible modeling language?}{LP / QP / conic / DCP --- not strictly needed}
  \begin{columns}[c]
    \begin{column}{.58\textwidth}
      \begin{itemize}
      \item Canonical forms ($A, b, c$, cone description) $\Rightarrow$ \alert{one-shot CPU$\to$GPU copy} of problem data
      \item Solver \alert{can iterate entirely on the device}
      \item Solution copied \alert{once back} to CPU
      \item These transfers can be a \alert{significant overhead}, but the cost is \alert{amortized} over many solver iterations
      \end{itemize}
    \end{column}
    \begin{column}{.40\textwidth}

\begin{frame}{Why a GPU-compatible modeling language?}{NLP --- the picture is different}
  \begin{columns}[c]
    \begin{column}{.58\textwidth}
      \begin{itemize}\itemsep2pt
      \item Solver \alert{queries the modeling layer at every iteration} for
        $f(x),\; g(x),\; \nabla f(x),\; \nabla g(x),\; \nabla^2_{xx} L(x,\lambda)$
      \item Each query depends on the current iterate---\alert{no one-shot transfer}
      \item Without GPU-native callbacks, every iteration pays a CPU$\leftrightarrow$GPU round-trip: \alert{significant per-iteration overhead, no amortization}
      \item<2-> JuMP today does \alert{not} emit GPU-native callbacks. Fine: modeling infrastructure stays on CPU, \alert{as long as the callbacks themselves run on the device}
      \end{itemize}
    \end{column}
    \begin{column}{.40\textwidth}
      \centering
      \begin{tikzpicture}[every node/.style={font=\scriptsize, align=center}]

\section{The ExaModels JuMP Interface}
% =============================================================
\begin{frame}[fragile]{The ExaModels JuMP Interface}{Status Quo}
  \scriptsize
  \begin{columns}[T]
    \begin{column}{.49\textwidth}
      \alert{\bfseries JuMP $\to$ ExaModels (via the MOI bridge)}
\begin{lstlisting}
using JuMP, ExaModels, CUDA, MadNLPGPU

N = 100
jm = Model()
@variable(jm, x[i=1:N],
    start = isodd(i) ? -1.2 : 1.0)
@objective(jm, Min, sum(100*(x[i+1] - x[i]^2)^2
                       + (1 - x[i])^2 for i = 1:N-1))

em = ExaModel(jm; backend = CUDABackend())
result = madnlp(em)
\end{lstlisting}
    \end{column}
    \begin{column}{.49\textwidth}
      \alert{\bfseries Native ExaModels}
\begin{lstlisting}
using ExaModels, CUDA, MadNLPGPU

N = 100
core = ExaCore(backend = CUDABackend())
@add_var(core, x, N;
    start = (isodd(i) ? -1.2 : 1.0 for i = 1:N))
@add_obj(core, 100*(x[i+1] - x[i]^2)^2
              + (1 - x[i])^2 for i = 1:N-1)
em = ExaModel(core); madnlp(em)
\end{lstlisting}
    \end{column}
  \end{columns}
  \smallskip
  \scriptsize
  Bridge alternative: \texttt{set\_optimizer(jm, ExaModels.Optimizer(...))}. Bridge supports \texttt{@constraint} / \texttt{@objective} only (\alert{not legacy \texttt{@NLconstraint} / \texttt{@NLobjective}}).
  \smallskip
  \visible<2->{
  \begin{block}{Model-build cost (Rosenbrock; this desktop)}
    \centering
    At $N=10^5$: JuMP route \alert{$\sim 1.74$\,s} vs Native ExaModels \alert{$\sim 0.44$\,ms} \;---\; \alert{$\sim 4000\times$ gap}, growing with $N$.
  \end{block}
  }
\end{frame}

\section{SIMD abstraction}
% =============================================================
\begin{frame}{An alternative to vectorization}
  \begin{itemize}
  \item Two strategies preserve indexed-constraint structure at the modeling layer:
  \end{itemize}
  \begin{columns}[t]
    \begin{column}{.48\textwidth}
      \alert{\bfseries Vectorization}
      \begin{itemize}
      \item Express each indexed family as a single matrix-algebra operation
      \item Canonical: CasADi MX, NumPy-style arrays
      \item Heterogeneous indexed structure (e.g., ACOPF power balance) requires \alert{pre-built sparse incidence matrices}
      \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
      \visible<2->{
      \alert{\bfseries Templatizing}
      \begin{itemize}
      \item Parameterize a single expression over a \alert{data iterator}; evaluate across instances
      \item Canonical: Gravity, ExaModels
      \item Heterogeneous structure handled \alert{directly by the iterator}
      \end{itemize}
      }
    \end{column}
  \end{columns}
  \medskip
  \visible<3->{
    \begin{block}{ExaModels' SIMD abstraction}
      Each objective term / constraint = a \alert{(pattern, data iterator)} pair. One compiled derivative kernel per pattern, executed across all data points in SIMD fashion.
    \end{block}
  }
\end{frame}

\begin{frame}[fragile]{Goddard rocket in ExaModels}
\begin{lstlisting}
using ExaModels, NLPModelsIpopt

core = ExaCore(minimize = false)                      # maximize final altitude
@add_var(core, h, 0:nh; start = 1.0, lvar = 1.0)
@add_var(core, v, 0:nh; start = (i/nh*(1-i/nh) for i=0:nh))
@add_var(core, m, 0:nh; start = ..., lvar = m_f, uvar = m_0)
@add_var(core, T, 0:nh; start = T_max/2, lvar = 0.0, uvar = T_max)
@add_var(core, step, 1; start = 1/nh, lvar = 0.0)

@add_obj(core, h[nh])                                 # objective: final altitude
@add_con(core, c1, -h[i]+h[i-1] + 0.5*step[1]*(v[i]+v[i-1]) for i=1:nh)
@add_con(core, c2, ...)                               # velocity dynamics
@add_con(core, c3, ...)                               # mass dynamics
@add_con(core, h[0] - h_0); @add_con(core, v[0] - v_0); ...   # boundary conditions

result = ipopt(ExaModel(core))                        # CPU, via NLPModels
\end{lstlisting}
  \begin{itemize}
  \item<2-> Three dynamics + four boundaries = \alert{5 patterns}, regardless of \texttt{nh}
  \item<3-> Same code on GPU: change \alert{one line} ---
        \texttt{core = ExaCore(minimize = false, backend = CUDABackend())}
  \end{itemize}
\end{frame}

\section{Stay in JuMP: GenOpt.jl}
% =============================================================
\begin{frame}[fragile]{GenOpt.jl \citep{legatGenOptJl2025} --- preserve patterns at JuMP-side construction}
  \texttt{GenOpt.jl} (Benoît Legat) packs ``groups of similar constraints / sums of similar terms'' into \alert{lightweight generators} that ExaModels can recognize.
\begin{lstlisting}
using GenOpt, JuMP, CUDA, MadNLP

@constraint(model, [i in 1:n], x[1,i] == x0[i],
    container = ParametrizedArray)                            # grouped constraint

@objective(model, Min,
    lazy_sum(0.5 * R[j] * u[i,j]^2 for i in 1:N, j in 1:p))   # grouped sum

set_optimizer(model, () -> GenOpt.ExaOptimizer(madnlp, CUDABackend()))
optimize!(model)
\end{lstlisting}
  \begin{itemize}
  \item<2-> Patterns are preserved at \alert{JuMP-side construction}---ExaModels receives them \alert{intact} (no scalar enumeration)
  \item<3-> Minimal model rewrite: add the \texttt{container} keyword / wrap sums in \texttt{lazy\_sum}
  \item<4-> Originally presented at JuMP-dev 2025
  \end{itemize}
  \refbr{
    \fullcite{legatGenOptJl2025}
  }
\end{frame}

\section{Under the hood}
% =============================================================
\begin{frame}{Type-stable expression tree + coloring-free sparse AD}
  \alert{\bfseries Parameterized expression tree}
  \begin{itemize}
  \item Each pattern compiles to a \alert{type-stable} graph; Julia compiler specializes a dedicated evaluator per pattern
  \item<2-> Same tree, different data and arguments, \alert{no recompilation}
  \end{itemize}
  \medskip
  \visible<3->{
  \alert{\bfseries Coloring-free sparse AD}
  \begin{itemize}
  \item Standard sparse reverse-mode AD uses \alert{graph coloring}---GPU-unfriendly (irregular control flow, dynamic dispatch)
  \item<4-> ExaModels: \alert{a priori} sparsity analysis per pattern $\Rightarrow$ sparse Jacobian + Lagrangian Hessian in \alert{COO directly, no coloring round}
  \end{itemize}
  }
\end{frame}

\section{GPU execution}
% =============================================================
\begin{frame}{One model, six backends}
  \begin{itemize}
  \item \alert{SIMDFunction} is the compiled artifact for one pattern: a per-data-point kernel + sparsity map
  \item<2-> CPU evaluation: \texttt{@simd} loop over data points
  \item<3-> GPU evaluation: launched via \alert{KernelAbstractions.jl} \citep{churavyKernelAbstractionsjl2025}
  \item<4-> Same model code targets:
    \begin{itemize}
    \item \texttt{CPU()} (multi-threaded) \quad $\bullet$\quad \texttt{CUDABackend()} (NVIDIA) \quad $\bullet$\quad \texttt{ROCBackend()} (AMD)
    \item \texttt{OneAPIBackend()} (Intel) \quad $\bullet$\quad \texttt{MetalBackend()} (Apple) \quad $\bullet$\quad \texttt{OpenCLBackend()} (generic)
    \end{itemize}
  \item<5-> Implementation choices that matter on GPU:
    \begin{itemize}
    \item COO (not CSC) for sparse output --- coalesced writes
    \item Objective: per-pattern buffer + reduction
    \item Gradient: sparse accumulation + compress
    \end{itemize}
  \item<6-> Currently the GPU IPM consumer is \alert{MadNLP + MadNLPGPU} \citep{shinMadNLPjl2025,pacaudCondensedspaceMethodsNonlinear2024}
  \end{itemize}
  \refbr{
    \fullcite{churavyKernelAbstractionsjl2025}\\
    \fullcite{shinMadNLPjl2025}
  }
\end{frame}

  \end{itemize}
  \smallskip
  \alert{\bfseries Open}
  \begin{itemize}\itemsep1pt
  \item<3-> \texttt{@NLconstraint} / \texttt{@NLobjective} not supported by the bridge
  \item<4-> Heterogeneous indexed structure does not round-trip $\Rightarrow$ use GenOpt.jl or native
  \item<5-> OpenCL / oneAPI backends held by a GPUCompiler pin (upstream SPIR-V issue)
  \end{itemize}
\end{frame}
