\section{What is ExaModels?}
% =============================================================
\begin{frame}[fragile]{What is ExaModels?}
  \begin{columns}[T]
    \begin{column}{.49\textwidth}
      \begin{itemize}\itemsep3pt
      \item \alert{Algebraic modeling system} embedded in Julia
      \item Same philosophy as JuMP / AMPL / CasADi / Pyomo: a \alert{convenient indexed syntax} for the user
      \item Uses \alert{scalar-based indexing} (vs vectorization \`a la CasADi MX) --- like JuMP
      \item Builds an \texttt{NLPModels.AbstractNLPModel}: feeds Ipopt, KNITRO, MadNLP, Uno, \dots
      \item<2-> \alert{\bfseries Killer feature}: GPU-native model + derivative evaluation; same code, change \alert{one keyword} (\texttt{backend = CUDABackend()}); six KA.jl backends
      \end{itemize}
    \end{column}
    \begin{column}{.49\textwidth}
      {\scriptsize\itshape Example: Goddard rocket --- maximize final altitude.\par}
      \lstinputlisting[basicstyle=\ttfamily\tiny]{code/03_what_is_snippet.jl}
    \end{column}
  \end{columns}
\end{frame}

\section{Why GPU-native modeling, only for NLPs?}
% =============================================================
\begin{frame}{Why a GPU-compatible modeling language?}{LP / QP / conic / DCP --- not strictly needed}
  \begin{columns}[c]
    \begin{column}{.58\textwidth}
      \begin{itemize}
      \item Canonical forms ($A, b, c$, cone description) $\Rightarrow$ \alert{one-shot CPU$\to$GPU copy} of problem data
      \item Solver \alert{can iterate entirely on the device}
      \item Solution copied \alert{once back} to CPU
      \item These transfers can be a \alert{significant overhead}, but the cost is \alert{amortized} over many solver iterations
      \end{itemize}
    \end{column}
    \begin{column}{.40\textwidth}
      \centering
      \begin{tikzpicture}[every node/.style={font=\scriptsize, align=center}]
        % CPU image
        \node (cpu) at (0, 2.7) {\includegraphics[width=1.6cm]{cpu_chip_parisc.jpg}};
        \node[font=\bfseries, color=mitdarkgray] at (0, 3.55) {CPU};
        % GPU image
        \node (gpu) at (0, 0) {\includegraphics[width=2.0cm]{exa_a100_gpu.png}};
        \node[font=\bfseries, color=mitred] at (0, -0.95) {GPU};
        % arrows
        \draw[->, very thick, mitred] (-0.45, 2.05) -- (-0.45, 0.8);
        \node[font=\tiny, color=mitdarkgray, anchor=east, align=right] at (-0.5, 1.4) {problem\\(once)};
        \draw[->, very thick, mitred] (0.45, 0.8) -- (0.45, 2.05);
        \node[font=\tiny, color=mitdarkgray, anchor=west, align=left] at (0.5, 1.4) {solution\\(once)};
        % PCIe label
        \node[font=\tiny\bfseries, color=mitdarkgray, fill=white, inner sep=2pt] at (0, 1.4) {PCIe / SXM};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Why a GPU-compatible modeling language?}{NLP --- the picture is different}
  \begin{columns}[c]
    \begin{column}{.58\textwidth}
      \begin{itemize}\itemsep2pt
      \item Solver \alert{queries the modeling layer at every iteration} for
        $f(x),\; g(x),\; \nabla f(x),\; \nabla g(x),\; \nabla^2_{xx} L(x,\lambda)$
      \item Each query depends on the current iterate---\alert{no one-shot transfer}
      \item Without GPU-native callbacks, every iteration pays a CPU$\leftrightarrow$GPU round-trip: \alert{significant per-iteration overhead, no amortization}
      \item<2-> JuMP today does \alert{not} emit GPU-native callbacks. Fine: modeling infrastructure stays on CPU, \alert{as long as the callbacks themselves run on the device}
      \end{itemize}
    \end{column}
    \begin{column}{.40\textwidth}
      \centering

\section{Stay in JuMP: GenOpt.jl}
% =============================================================
\begin{frame}[fragile]{GenOpt.jl \citep{legatGenOptJl2025} --- preserve patterns at JuMP-side construction}
  \texttt{GenOpt.jl} (Benoît Legat) packs ``groups of similar constraints / sums of similar terms'' into \alert{lightweight generators} that ExaModels can recognize.
\begin{lstlisting}
using GenOpt, JuMP, CUDA, MadNLP

@constraint(model, [i in 1:n], x[1,i] == x0[i],
    container = ParametrizedArray)                            # grouped constraint

@objective(model, Min,
    lazy_sum(0.5 * R[j] * u[i,j]^2 for i in 1:N, j in 1:p))   # grouped sum

set_optimizer(model, () -> GenOpt.ExaOptimizer(madnlp, CUDABackend()))
optimize!(model)
\end{lstlisting}
  \begin{itemize}
  \item<2-> Patterns are preserved at \alert{JuMP-side construction}---ExaModels receives them \alert{intact} (no scalar enumeration)
  \item<3-> Minimal model rewrite: add the \texttt{container} keyword / wrap sums in \texttt{lazy\_sum}
  \item<4-> \emph{Legat's work, not mine} --- but a really cool proof-of-concept route. I'm thinking about something else next.
  \end{itemize}
  \refbr{
    \fullcite{legatGenOptJl2025}
  }
\end{frame}

\section{Other nice things about ExaModels}
% =============================================================
\begin{frame}{Other nice things about ExaModels}
  \begin{itemize}\itemsep4pt
  \item \alert{\bfseries Portability}: same model code on \texttt{CPU()}, \texttt{CUDABackend()}, \texttt{ROCBackend()}, \texttt{OneAPIBackend()}, \texttt{MetalBackend()}, \texttt{OpenCLBackend()} --- one backend keyword switches hardware.
  \item<2-> \alert{\bfseries Performance}: type-stable end-to-end $\Rightarrow$ Julia compiler inlines AD code per pattern; matches or beats AMPL / JuMP on CPU before any GPU is involved.
  \item<3-> \alert{\bfseries AOT compatibility}: works under \texttt{juliac --trim=safe} for standalone binaries (no runtime codegen in the hot path) --- demonstrated by \texttt{NLPModelsIpoptLite.jl}.
  \item<4-> \alert{\bfseries Two-stage + batch}: \texttt{TwoStageExaCore} for first-stage / recourse-stage structure across scenarios; \texttt{BatchExaCore} for many independent NLP instances sharing a structure --- natural fits for stochastic programming and parameter sweeps.
  \end{itemize}
\end{frame}

\section{Conclusion}
% =============================================================
\begin{frame}{Summary}
  \begin{center}
    \url{https://github.com/exanauts/ExaModels.jl}
  \end{center}
  \vspace{1em}
  \begin{itemize}
  \item ExaModels offers a \alert{SIMD abstraction} for NLPs---an alternative to vectorization that preserves indexed structure through (pattern, data iterator) pairs
  \item<2-> Type-stable parameterized expression tree + coloring-free sparse AD make \alert{GPU-native} derivative evaluation practical
  \item<3-> One model, six KA.jl backends, no code change
  \item<4-> Already the modeling layer for MadNLP's GPU-native interior-point methods \citep{pacaudCondensedspaceMethodsNonlinear2024}
  \end{itemize}
  \medskip
  \visible<5->{
    \begin{block}{Open question --- and continuing this thread}
      \scriptsize We don't need to move all of JuMP onto the GPU. The win is \alert{GPU-native callbacks} eliminating the per-iteration round-trip. \texttt{GenOpt.jl} is a proof-of-concept; \alert{deeper opportunity}: JuMP stores expressions in \alert{non-concrete} types, so the conversion loop pays per-iteration type inference. \alert{Concretely typed expression trees in JuMP would make the bridge cheap.}\\[2pt]
      Continuing at \alert{SIAM Optimization 2026}: ``The State of ExaModels'' (MS369), \alert{Fri June 5, 11:10 BST}, 50 George Square G.03.
    \end{block}
  }
\end{frame}
