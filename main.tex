\begin{frame}[fragile]{What is ExaModels?}
  \begin{columns}[T]
    \begin{column}{.48\textwidth}
      \begin{itemize}\itemsep4pt
      \item \alert{Algebraic modeling system} in Julia, JuMP-inspired syntax.
      \item<2-> Builds on \texttt{NLPModels.jl} --- compatible with: \texttt{Ipopt}, \texttt{MadNLP}, \texttt{KNITRO}, \texttt{Uno}, \dots (essentially all JSO-style solvers)
      \item<3-> Ships with its own AD system --- $f(x),\; g(x),\; \nabla_{x} f(x),\; \nabla_{x} g(x),\; \nabla^2_{xx} L(x,\lambda)$
      \item<4-> \alert{\bfseries Killer feature}: \alert{GPU-compatible modeling} --- more on this later.
      \item<5-> \alert{\bfseries Design philosophy}: not optimized for ease-of-use, optimized for \alert{runtime performance}. Aim: the \alert{fastest modeling system} for building and running optimization solves.
      \end{itemize}
    \end{column}

\section{Why GPU-native modeling, only for NLPs?}
% =============================================================
\begin{frame}{Why a GPU-compatible modeling language?}
  \framesubtitle{%
    \only<1>{LP / QP / DCP --- not strictly needed}%
    \only<2->{NLP --- the picture is different}%
  }
  \begin{columns}[T]
    \begin{column}{.55\textwidth}
      \only<1>{%
        \begin{itemize}
        \item Canonical forms ($A, b, c$, ...) $\Rightarrow$ \alert{one-shot CPU$\to$GPU copy} of problem data
        \item Transfers can be an overhead, but the cost is \alert{amortized} over many solver iterations
        \item \alert{Existing GPU-native solvers} in this class already work this way: \texttt{CuClarabel}, \texttt{cuPDLP+}
        \item[] $\Rightarrow$ For LP / QP / DCP, a \alert{GPU-native modeling language may not add much value}
        \end{itemize}
      }%
      \only<2->{%
        \begin{itemize}\itemsep2pt
        \item Solver \alert{queries the modeling layer at every iteration} via \alert{callbacks} ---
          $f(x),\; g(x),\; \nabla_{x} f(x),\; \nabla_{x} g(x),\; \nabla^2_{xx} L(x,\lambda)$
        \item Each query depends on the current iterate --- \alert{no one-shot transfer}
        \item Without GPU-native callbacks, every iteration pays a CPU$\leftrightarrow$GPU round-trip: \alert{significant per-iteration overhead, no amortization}
        \item<3-> JuMP today does \alert{not} emit GPU-native callbacks. Fine: modeling infrastructure stays on CPU, \alert{as long as the callbacks themselves run on the device}
        \end{itemize}
      }%
    \end{column}
    \begin{column}{.43\textwidth}
      \centering

using GenOpt, JuMP, CUDA, MadNLP

model = Model()
@constraint(model, [i in 1:n], x[1,i] == x0[i], container = ParametrizedArray) # grouped constraint
@objective(model, Min, lazy_sum(0.5 * R[j] * u[i,j]^2 for i in 1:N, j in 1:p)) # grouped sum

set_optimizer(model, () -> GenOpt.ExaOptimizer(madnlp, CUDABackend()))
optimize!(model)
\end{lstlisting}
\end{codebox}

  \item Uses JuMP's own \alert{extension hooks} so the algebraic patterns \alert{never get unrolled} --- ExaModels sees the structured representation directly
  \item<2-> Minimal model rewrite: \texttt{container = ParametrizedArray} on \texttt{@constraint}, \texttt{lazy\_sum} on \texttt{@objective}
  \item<3-> A really cool proof-of-concept route --- I'm thinking about something else next.
  \end{itemize}

\section{Alternative: ExaModels as a JuMP backend for GPU}
% =============================================================
\begin{frame}{Alternative: ExaModels as a JuMP backend for GPU acceleration}{Idea: a small number of fully-typed buckets}
  \begin{itemize}\itemsep2pt
  \item Don't ask JuMP to be fully type-stable. Give it a \alert{small number of buckets} --- each potentially large --- whose contents are \alert{fully typed and concrete}
  \item<2-> $\sim$10 patterns of practical significance: \alert{linear}, \alert{quadratic}, \alert{bilinear}, \alert{poly $\times\,\sin$}, \alert{poly $\times\,\exp$}, \dots, plus an \alert{Any fallback} bucket for anything else
  \end{itemize}

\section{Conclusion}
% =============================================================
\begin{frame}{Summary}
  \begin{center}
    \url{https://github.com/exanauts/ExaModels.jl}
  \end{center}
  \smallskip
  \small
  \begin{itemize}\itemsep2pt
  \item ExaModels offers a \alert{SIMD abstraction} for NLPs --- pattern + data iterator
  \item<2-> Encoding the algebraic structure in the type buys us \alert{GPU-native callbacks}, portability across GPU backends, and \alert{AOT compilation} to a standalone binary
  \item<3-> The trade-off: \alert{all the burden goes onto the compiler} --- great when patterns are stable (specialized, branch-free code), expensive when they change (compile-once per pattern)
  \end{itemize}
  \smallskip
  \visible<4->{
    \begin{block}{Discussion}
      \small We don't need to move all of JuMP onto the GPU. The win is \alert{GPU-native callbacks} eliminating the per-iteration round-trip. A slightly more type-stable JuMP tree would make the bridge cheap.
    \end{block}
  }
