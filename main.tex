\section{The question}
% =============================================================
\begin{frame}{You have a JuMP model. You want it on a GPU.}
  \begin{itemize}
  \item JuMP gives you a clean modeling layer, a mature solver ecosystem, and CPU performance that scales to large problems
  \item<2-> GPUs sit one step further: \alert{factorization-based} optimization is starting to land on GPUs, with $\geq$10$\times$ speedups on the right problems \citep{pacaudCondensedspaceMethodsNonlinear2024}
  \item<3-> But solver speedup is wasted if \alert{the model layer} can't feed it---callbacks must run on the device
  \item<4-> Today's question:
  \end{itemize}
  \medskip
  \visible<5->{
    \begin{center}
      \it Can ExaModels power JuMP on GPUs?
    \end{center}
  }
  \refbr{
    \fullcite{pacaudCondensedspaceMethodsNonlinear2024}
  }
\end{frame}

% =============================================================
\section{The translation layer}
% =============================================================
\begin{frame}[fragile]{Two paths from JuMP to ExaModels}
  Load \texttt{ExaModels} + \texttt{JuMP} together---the \texttt{ExaModelsJuMP} / \texttt{ExaModelsMOI} package extensions activate automatically.
  \medskip
  \begin{onlyenv}<1->
  \alert{\bfseries Path 1 --- Direct conversion to an \texttt{ExaModel}}
\begin{lstlisting}
using ExaModels, JuMP, CUDA, MadNLPGPU

em = ExaModel(jm; backend = CUDABackend())   # JuMP -> GPU-ready ExaModel
result = madnlp(em)                          # solve via NLPModels API
\end{lstlisting}
  \end{onlyenv}
  \medskip
  \begin{onlyenv}<2->
  \alert{\bfseries Path 2 --- ExaModels as a JuMP optimizer}
\begin{lstlisting}
using ExaModels, JuMP, CUDA, MadNLP, MadNLPGPU

set_optimizer(jm, () -> ExaModels.Optimizer(MadNLP.madnlp, CUDABackend()))
optimize!(jm)                                # stay in JuMP, solve on GPU
\end{lstlisting}
  \end{onlyenv}
  \medskip
  \begin{itemize}
  \item<3-> Both routes go through the same MOI bridge: \alert{recover patterns} from the scalar collection by \alert{bucketing matching expression heads} into templatized sums
  \end{itemize}
\end{frame}

\begin{frame}{Where the bridge breaks}
  \begin{itemize}
  \item JuMP lowers \texttt{@constraint(model, [i=1:N], ...)} to a flat scalar collection \alert{before} the model is submitted to MOI
  \item<2-> Pattern information visible to the user is \alert{no longer visible} to MOI---the bridge recovers patterns only when scalar constraints happen to share an expression head
  \item<3-> Heterogeneous indexed structure (e.g., variable-size sums per index, as in ACOPF power balance) does \alert{not} round-trip
  \item<4-> Caveat for legacy JuMP code: only \texttt{@constraint} / \texttt{@objective} are supported; \alert{\texttt{@NLconstraint} / \texttt{@NLobjective} are not}
  \end{itemize}
  \medskip
  \visible<5->{
    \begin{center}
      \it To get the full speedup we need to keep the pattern alive through model construction.
    \end{center}
  }
\end{frame}

% =============================================================
\section{Why patterns matter}
% =============================================================
\begin{frame}{Large-scale NLPs have repetitive structure}
  \begin{itemize}
  \item NLPs in engineering / science / statistics routinely involve millions of variables and constraints
  \item<2-> Yet their formulations fit on a few pages---the same \alert{algebraic pattern} repeats across spatial nodes, time steps, scenarios, data samples
  \item<3-> ACOPF on a network with thousands of buses: only \alert{15 distinct algebraic patterns} \citep{shinAcceleratingOptimalPower2024}
  \item<4-> Problem size is driven by the number of \alert{data points}, not the number of patterns
  \item<5-> Most AMSs (AMPL, GAMS, JuMP) flatten this at construction time. Cost: redundant work, no SIMD parallelism over patterns, \alert{no path to native GPU evaluation}
  \end{itemize}
  \refbr{
    \fullcite{shinAcceleratingOptimalPower2024}
  }
\end{frame}

% =============================================================
\section{SIMD abstraction}
% =============================================================
\begin{frame}{An alternative to vectorization}
  \begin{itemize}
  \item Two strategies preserve indexed-constraint structure at the modeling layer:
  \end{itemize}
  \begin{columns}[t]
    \begin{column}{.48\textwidth}
      \alert{\bfseries Vectorization}
      \begin{itemize}
      \item Express each indexed family as a single matrix-algebra operation
      \item Canonical: CasADi MX, NumPy-style arrays
      \item Heterogeneous indexed structure (e.g., ACOPF power balance) requires \alert{pre-built sparse incidence matrices}
      \end{itemize}
    \end{column}
    \begin{column}{.48\textwidth}
      \visible<2->{
      \alert{\bfseries Templatizing}
      \begin{itemize}
      \item Parameterize a single expression over a \alert{data iterator}; evaluate across instances
      \item Canonical: Gravity, ExaModels
      \item Heterogeneous structure handled \alert{directly by the iterator}
      \end{itemize}
      }
    \end{column}
  \end{columns}
  \medskip
  \visible<3->{
    \begin{block}{ExaModels' SIMD abstraction}
      Each objective term / constraint = a \alert{(pattern, data iterator)} pair. One compiled derivative kernel per pattern, executed across all data points in SIMD fashion.
    \end{block}
  }
\end{frame}

\begin{frame}{Pattern--data separation, concretely}
  \begin{itemize}
  \item \alert{Pattern}: a single algebraic expression with free indices into variables, parameters, and a data point
  \item<2-> \alert{Data iterator}: the collection of data points over which the pattern is evaluated
  \item<3-> Model is stored as a small collection of \texttt{SIMDFunction}s---one per pattern---together with their data
  \item<4-> Each \texttt{SIMDFunction} bundles: (i) the parameterized expression tree, (ii) compressor maps for the Jacobian/Hessian, (iii) offsets into the global output arrays
  \item<5-> Constructed once at model compilation, immutable thereafter; updating $x$ or $\theta$ requires \alert{no recompilation}
  \end{itemize}
\end{frame}

\begin{frame}{Worked example: Goddard rocket \citep{goddardMethodReachingExtreme1919}}
  \small
  \begin{align*}
    \max\;& h_T \\
    \text{s.t.}\;& h_t = h_{t-1} + \tfrac{\Delta t}{2}(v_t + v_{t-1}), && t=1,\dots,T \\
    & v_t = v_{t-1} + \tfrac{\Delta t}{2}\!\left(\tfrac{\tau_t - D(h_t,v_t)}{m_t} - g(h_t) + \tfrac{\tau_{t-1} - D(h_{t-1},v_{t-1})}{m_{t-1}} - g(h_{t-1})\right), && t=1,\dots,T \\
    & m_t = m_{t-1} - \tfrac{\Delta t}{2c}(\tau_t + \tau_{t-1}), && t=1,\dots,T \\
    & h_0 = h^0,\; v_0 = 0,\; m_0 = m^0,\; m_T = m^f
  \end{align*}
  \begin{itemize}
  \item<2-> Three dynamics constraints + boundary conditions = \alert{5 patterns}, regardless of $T$
  \item<3-> As $T$ grows from $10^2$ to $10^6$, pattern count is \alert{fixed}; data-point count scales
  \item<4-> Refining the discretization is a \alert{data} change, not a \alert{model} change
  \end{itemize}
  \refbr{
    \fullcite{goddardMethodReachingExtreme1919}\\
    \fullcite{dolanBenchmarkingOptimizationSoftware2001}
  }
\end{frame}

\begin{frame}{Where ExaModels sits among AMSs with their own AD}
  \begin{center}\small
    \begin{tabular}{lllc}
      \toprule
      System & Language & AD strategy & Pattern-aware \\
      \midrule
      AMPL    & own            & ASL (built-in)            & no \\
      GAMS    & own            & built-in                  & no \\
      JuMP    & Julia          & reverse-mode on graph     & no \\
      CasADi  & C++/Py/MATLAB  & source transform          & partial \\
      GEKKO   & Python         & APMonitor backend         & no \\
      Gravity & C++            & symbolic                  & yes \\
      \midrule
      \alert{ExaModels.jl} & Julia & \alert{type-stable per-pattern} & \alert{yes} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
  \item<2-> Two strategies preserve indexed structure: \alert{vectorization} (CasADi MX) and \alert{templatizing} (Gravity, ExaModels)
  \item<3-> ExaModels = templatizing + Julia type-stable graph + KernelAbstractions.jl GPU portability
  \item<4-> Neither CasADi nor Gravity supports GPU-native derivative evaluation
  \end{itemize}
\end{frame}

\section{Two routes for JuMP users}
% =============================================================
\begin{frame}{Route A: native ExaModels API}
  Five-step workflow:
  \begin{enumerate}
  \item \texttt{ExaCore()} --- a builder that progressively accumulates model information
  \item<2-> \texttt{@add\_var(c, x, dims; start, lvar, uvar)} --- variable block with bounds and initial point
  \item<3-> \texttt{@add\_obj(c, expr for i in data)} --- objective pattern + data iterator
  \item<4-> \texttt{@add\_con(c, name, expr for i in data; lcon, ucon)} --- constraint pattern (named blocks for accessors)
  \item<5-> \texttt{ExaModel(c)} --- compile to an \texttt{NLPModels.AbstractNLPModel}
  \end{enumerate}
  \medskip
  \begin{itemize}
  \item<6-> Also: \texttt{@add\_par} (parameters), \texttt{@add\_con!} (augment an existing constraint block), \texttt{@add\_expr} (shared subexpressions), \texttt{@register\_univariate}/\texttt{@register\_bivariate} (custom functions)
  \item<7-> Consumable by Ipopt, KNITRO, MadNLP, Uno, Percival, DCISolver, FletcherPenaltySolver
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Goddard rocket in ExaModels}
\begin{lstlisting}
using ExaModels, NLPModelsIpopt

core = ExaCore(minimize = false)                      # maximize final altitude
@add_var(core, h, 0:nh; start = 1.0, lvar = 1.0)
@add_var(core, v, 0:nh; start = (i/nh*(1-i/nh) for i=0:nh))
@add_var(core, m, 0:nh; start = ..., lvar = m_f, uvar = m_0)
@add_var(core, T, 0:nh; start = T_max/2, lvar = 0.0, uvar = T_max)
@add_var(core, step, 1; start = 1/nh, lvar = 0.0)

@add_obj(core, h[nh])                                 # objective: final altitude
@add_con(core, c1, -h[i]+h[i-1] + 0.5*step[1]*(v[i]+v[i-1]) for i=1:nh)
@add_con(core, c2, ...)                               # velocity dynamics
@add_con(core, c3, ...)                               # mass dynamics
@add_con(core, h[0] - h_0); @add_con(core, v[0] - v_0); ...   # boundary conditions

result = ipopt(ExaModel(core))                        # CPU, via NLPModels
\end{lstlisting}
  \begin{itemize}
  \item<2-> Three dynamics + four boundaries = \alert{5 patterns}, regardless of \texttt{nh}
  \item<3-> Same code on GPU: change \alert{one line} ---
        \texttt{core = ExaCore(minimize = false, backend = CUDABackend())}
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Route B: GenOpt.jl \citep{legatGenOptJl2025} --- stay in JuMP, keep patterns}
  \texttt{GenOpt.jl} (Benoît Legat) packs ``groups of similar constraints / sums of similar terms'' into \alert{lightweight generators} that ExaModels can recognize.
\begin{lstlisting}
using GenOpt, JuMP, CUDA, MadNLP

@constraint(model, [i in 1:n], x[1,i] == x0[i],
    container = ParametrizedArray)                            # grouped constraint

@objective(model, Min,
    lazy_sum(0.5 * R[j] * u[i,j]^2 for i in 1:N, j in 1:p))   # grouped sum

set_optimizer(model, () -> GenOpt.ExaOptimizer(madnlp, CUDABackend()))
optimize!(model)
\end{lstlisting}
  \begin{itemize}
  \item<2-> Patterns are preserved at \alert{JuMP-side construction}---ExaModels receives them \alert{intact} (no scalar enumeration)
  \item<3-> Minimal model rewrite: add the \texttt{container} keyword / wrap sums in \texttt{lazy\_sum}
  \item<4-> Originally presented at JuMP-dev 2025
  \end{itemize}
  \refbr{
    \fullcite{legatGenOptJl2025}
  }
\end{frame}

\section{Under the hood}
% =============================================================
\begin{frame}{Parameterized expression tree}
  \begin{itemize}
  \item Each pattern compiles to a \alert{type-stable} expression tree
  \item<2-> Node types: \texttt{Var}, \texttt{DataIndexed}, \texttt{ParameterNode}, \texttt{Constant\{T\}}, \texttt{Node1}, \texttt{Node2}
  \item<3-> Nested type parameterization $\Rightarrow$ Julia compiler specializes a dedicated evaluator per pattern
  \item<4-> The same tree is evaluated and differentiated with \alert{different data and arguments}, no recompilation
  \item<5-> Algebraic simplification at compile time via multiple dispatch
  \end{itemize}
\end{frame}

\begin{frame}{Coloring-free sparse AD}
  \begin{itemize}
  \item Standard sparse reverse-mode AD requires \alert{graph coloring} to reuse adjoint passes across nonzero columns
  \item<2-> Coloring is GPU-unfriendly: irregular control flow, dynamic dispatch
  \item<3-> ExaModels: \alert{a priori} sparsity analysis per pattern at model-construction time
    \begin{itemize}
    \item Relative index sequence of nonzeros within each pattern is compressed into a compact mapping
    \item Sparse Jacobian and Lagrangian Hessian in COO directly, no coloring round
    \end{itemize}
  \item<4-> First-order: adjoint tree (\texttt{drpass}, \texttt{grpass}). Second-order: second-adjoint tree (\texttt{hrpass}, \texttt{hdrpass}, \texttt{hrpass0})
  \end{itemize}
\end{frame}

% =============================================================
\section{GPU execution}
% =============================================================
\begin{frame}{One model, six backends}
  \begin{itemize}
  \item \alert{SIMDFunction} is the compiled artifact for one pattern: a per-data-point kernel + sparsity map
  \item<2-> CPU evaluation: \texttt{@simd} loop over data points
  \item<3-> GPU evaluation: launched via \alert{KernelAbstractions.jl} \citep{churavyKernelAbstractionsjl2025}
  \item<4-> Same model code targets:
    \begin{itemize}
    \item \texttt{CPU()} (multi-threaded) \quad $\bullet$\quad \texttt{CUDABackend()} (NVIDIA) \quad $\bullet$\quad \texttt{ROCBackend()} (AMD)
    \item \texttt{OneAPIBackend()} (Intel) \quad $\bullet$\quad \texttt{MetalBackend()} (Apple) \quad $\bullet$\quad \texttt{OpenCLBackend()} (generic)
    \end{itemize}
  \item<5-> Implementation choices that matter on GPU:
    \begin{itemize}
    \item COO (not CSC) for sparse output --- coalesced writes
    \item Objective: per-pattern buffer + reduction
    \item Gradient: sparse accumulation + compress
    \end{itemize}
  \item<6-> Currently the GPU IPM consumer is \alert{MadNLP + MadNLPGPU} \citep{shinMadNLPjl2025,pacaudCondensedspaceMethodsNonlinear2024}
  \end{itemize}
  \refbr{
    \fullcite{churavyKernelAbstractionsjl2025}\\
    \fullcite{shinMadNLPjl2025}
  }
\end{frame}

\section{Beyond JIT}
% =============================================================
\begin{frame}{Ahead-of-time compilation \& numeric precision}
  \begin{itemize}
  \item \alert{No runtime code generation in the hot path} --- the model evaluation surface is pure type-stable Julia
  \item<2-> Compatible with \texttt{juliac --trim=safe} for AOT binaries; \texttt{NLPModelsIpoptLite.jl} demonstrates the end-to-end path
  \item<3-> Same code paths serve JIT and AOT --- no separate codegen step (unlike CasADi)
  \item<4-> Generic numeric precision: Float32, Float64, \texttt{DoubleFloats}, user-defined types --- \texttt{replace\_T} swaps the scalar type
  \end{itemize}
\end{frame}

\begin{frame}{Extensions: parametric, two-stage stochastic, batch}
  \begin{itemize}
  \item \alert{Parametric problems}: \texttt{set\_value!} updates problem data without model reconstruction --- key for MPC / parameter sweeps
  \item<2-> \alert{Two-stage stochastic}: \texttt{TwoStageExaCore} + \texttt{EachScenario()}; first-stage variables shared, second-stage replicated across scenarios
  \item<3-> \alert{Batch models}: \texttt{BatchExaCore} stores independent NLP instances as columns of a matrix; \texttt{FlatNLPModel} exposes a single callback API to the solver
  \item<4-> All three reuse the same SIMD abstraction + AD machinery
  \end{itemize}
\end{frame}
